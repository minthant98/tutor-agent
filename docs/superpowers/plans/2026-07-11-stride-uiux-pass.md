## Global Constraints

Every task's requirements implicitly include this section. Copy binding rules verbatim into any subagent dispatch.

- **Analytical copy discipline** — never praise, never speculate, always explain *why*. Applies to all product copy, not only Alex-authored strings.
- **No gamification** — no streaks, XP, badges, achievements, level-ups, celebratory motion.
- **Alex memory guard** — every LLM prompt that speaks as Alex receives student-history context. Anti-hallucination clause required in every Alex prompt.
- **Readiness gradient** — `Cool Blue → Indigo → Lavender → Soft Gold → Amber`. Never green. Never animated.
- **One Primary button per section** — universal rule.
- **Typography split** — Geist Mono for scores, percentages, marks, paper codes, question IDs, M1/A1/B1, timers, keyboard shortcut chips. Geist Sans for everything else.
- **Type scale** — 11 / 12 / 14 / 16 / 20 / 24 / 32 / 40 only.
- **Radii** — Input/Button 8, Card 10, Dialog 14, Badge full-round.
- **Motion** — durations 120 / 220 / 320. Single ease token `cubic-bezier(.22, .61, .36, 1)`. Respect `prefers-reduced-motion` — durations → 0.
- **Surfaces** — 0 app / 1 card / 2 elevated / 3 dialog+drawer. Shadows only on Surface 3.
- **Educational tokens** — `Question`, `Answer`, `Hint`, `MarkScheme`, `TeacherFeedback`, `AlexFeedback`, `Readiness`. Used as domain primitives, never ad-hoc.
- **Primary color** — `#6268F2` (warm indigo).
- **Neutrals** — warm-tinted. Dark: `#0E0E11` / `#151519` / `#1C1C21`. Light: `#FCFCFD` / `#F8F8FA` / `#FFFFFF`.
- **Focus states** — 2px indigo ring + 2px offset. Always visible; never removed.
- **Skeletons** — pulse only, never shimmer.
- **Empty states** — never say "No data". Always planner-positive next action + CTA.
- **Focus return** — focus returns to trigger element on dialog/drawer close.
- **URL discipline** — same URLs for v2 and v3; page components branch on flag. No parallel `/dashboard-v3` routes.
- **Feature flag pattern** — `useFeatureFlag("<flag>", false)` (sub-project #3 pattern).
- **Retirement** — after 7 days at 100%, delete v2 code + flag in same PR.
- **Readiness display** — internal storage 1 decimal; display rounded integer.
- **A11y** — never color-only status. Use symbol + label. `axe-core` CI check per component.
- **KaTeX** for all math; formulas token-colored with MarkScheme.

```typescript
import { forwardRef } from "react";
import { Slot } from "@radix-ui/react-slot";
import { cva, type VariantProps } from "class-variance-authority";
import { cn } from "@/lib/utils";

const buttonVariants = cva(
  "inline-flex items-center justify-center rounded-input font-sans text-14 font-medium transition-colors duration-120 ease-standard disabled:opacity-50 disabled:pointer-events-none focus-visible:outline-none",
  {
    variants: {
      variant: {
        primary: "bg-[--primary] text-white hover:bg-[--primary-focus]",
        secondary: "bg-transparent border border-border-subtle text-white hover:bg-surface-1",
        ghost: "bg-transparent text-white/70 hover:text-white hover:bg-surface-1",
        destructive: "bg-transparent border border-[#D48A8A]/40 text-[#D48A8A] hover:bg-[#D48A8A]/10",
      },
      size: {
        sm: "h-8 px-3",
        md: "h-10 px-4",
        lg: "h-12 px-6",
      },
    },
    defaultVariants: { variant: "primary", size: "md" },
  }
);

export interface ButtonProps
  extends React.ButtonHTMLAttributes<HTMLButtonElement>,
    VariantProps<typeof buttonVariants> {
  asChild?: boolean;
}

export const Button = forwardRef<HTMLButtonElement, ButtonProps>(
  ({ className, variant, size, asChild = false, ...props }, ref) => {
    const Comp = asChild ? Slot : "button";
    return <Comp className={cn(buttonVariants({ variant, size }), className)} data-variant={variant} ref={ref} {...props} />;
  }
);
Button.displayName = "Button";
```

```python
@router.get("/search")
async def search(
    q: str,
    context: str | None = None,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
) -> list[dict]:
    q_lower = q.lower()
    results: list[dict] = []
    # Topics
    topic_rows = (await db.execute(
        select(SyllabusTopic).where(
            SyllabusTopic.subject == user.primary_subject,
            or_(
                func.lower(SyllabusTopic.topic_id).contains(q_lower),
                func.lower(SyllabusTopic.label).contains(q_lower),
            ),
        ).limit(10)
    )).scalars().all()
    for t in topic_rows:
        results.append({
            "id": f"topic:{t.topic_id}",
            "type": "topic",
            "label": t.label,
            "subtitle": t.subject_label,
            "href": f"/topics/{t.topic_id}",
            "topic_id": t.topic_id,
        })
    # Recent submissions
    sub_rows = (await db.execute(
        select(GradedUpload).where(
            GradedUpload.student_id == user.id,
            GradedUpload.status == "graded",
        ).order_by(GradedUpload.created_at.desc()).limit(5)
    )).scalars().all()
    for s in sub_rows:
        results.append({
            "id": f"submission:{s.id}",
            "type": "submission",
            "label": s.question_text[:80],
            "subtitle": f"{s.marks_awarded}/{s.max_marks}",
            "href": f"/mark/{s.id}",
            "topic_id": s.topic,
        })
    # Context-aware rerank
    if context:
        results.sort(key=lambda r: (0 if r.get("topic_id") == context else 1))
    return results
```

```typescript
"use client";
import { useEffect } from "react";

type Options = { scope?: "global" | "session"; ignoreInInput?: boolean };

export function useKeyboardShortcut(combo: string, handler: () => void, opts: Options = {}) {
  const { ignoreInInput = true } = opts;
  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      if (ignoreInInput && isTypingContext(e.target)) return;
      if (matches(e, combo)) { e.preventDefault(); handler(); }
    }
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [combo, handler, ignoreInInput]);
}

function isTypingContext(target: EventTarget | null): boolean {
  if (!(target instanceof HTMLElement)) return false;
  const tag = target.tagName;
  return tag === "INPUT" || tag === "TEXTAREA" || target.isContentEditable;
}

function matches(e: KeyboardEvent, combo: string): boolean {
  const parts = combo.toLowerCase().split("+");
  const key = parts.pop()!;
  const wantMeta = parts.includes("cmd") || parts.includes("ctrl");
  const wantShift = parts.includes("shift");
  return e.key.toLowerCase() === key
    && (!wantMeta || e.metaKey || e.ctrlKey)
    && (!wantShift || e.shiftKey);
}
```

```python
OBSERVATION_SYSTEM = """You are Alex generating up to 3 short observations about what a student did this week.

HARD RULES:
1. NEVER praise. Analytical only.
2. Every observation must be traceable to specific data. Evidence provided below.
3. Return JSON array of {"text": "...", "trace_ref": "<one of the provided evidence keys>"}
4. Maximum 3 items. If evidence is thin, return fewer.
5. No exclamation marks. No emoji. No student name."""

async def generate_for_week(db, student_id, subject, week_of) -> list[Observation]:
    evidence = await _gather_evidence(db, student_id, subject, week_of)
    if not evidence["items"]:
        return []
    response = await llm.generate_json(
        prompt=f"Evidence:\n{json.dumps(evidence, indent=2)}\n\nWrite up to 3 observations.",
        system=OBSERVATION_SYSTEM,
    )
    obs_list = response if isinstance(response, list) else response.get("observations", [])
    saved = []
    for item in obs_list[:3]:
        ref = item["trace_ref"]
        ev = evidence["items"].get(ref)
        if not ev:
            continue  # anti-hallucination: drop unreferenced items
        saved.append(Observation(
            student_id=student_id,
            subject=subject,
            text=item["text"],
            trace_json={"queries": ev["queries"], "session_ids": ev["session_ids"], "evidence_summary": ev["summary"]},
            week_of=week_of,
        ))
    db.add_all(saved)
    await db.flush()
    return saved
```

```typescript
export function useAutosave<T>({ sessionId, state, debounceMs = 300 }: {
  sessionId: string; state: T; debounceMs?: number;
}) {
  const stateRef = useRef(state);
  stateRef.current = state;
  useEffect(() => {
    const t = setTimeout(() => {
      fetch(`/api/v1/sessions/${sessionId}/state`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(stateRef.current),
      });
    }, debounceMs);
    return () => clearTimeout(t);
  }, [state, sessionId, debounceMs]);
}
```

- [ ] **Step 2: Implement**

```python
@dataclass
class TopicStats:
    mastery: float          # 0..1
    days_since_practice: int
    prereq_children: int
    exam_frequency: float   # 0..1 syllabus weight

def recency_weight(days: int) -> float:
    # More recent = closer to 1; long-forgotten = higher (needs revisit)
    return min(1.0 + math.log1p(days) / 3.0, 3.0)

def prerequisite_multiplier(children: int) -> float:
    return 1.0 + math.log1p(children) * 0.4

def impact_score(t: TopicStats) -> float:
    weakness = 1.0 - t.mastery
    return weakness * recency_weight(t.days_since_practice) * prerequisite_multiplier(t.prereq_children) * (t.exam_frequency + 0.05)
```
